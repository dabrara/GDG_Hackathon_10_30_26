#!/usr/bin/env bash
# PlainSight one-file installer for a DigitalOcean GPU Droplet (Ubuntu 22.04 / 24.04).
# It pulls the GitHub repo, installs Ollama + a Gemma model, runs the app as a service,
# and puts HTTPS + a password in front of it. Re-run it any time to pull the latest code.
#
#   sudo bash deploy-plainsight.sh
#
# Optional settings (put them before the command, e.g.  sudo MODEL=gemma4:e4b bash deploy-plainsight.sh):
#   APP_PASS      password for the site (random one is made and printed if you leave it off)
#   APP_USER      login name (default: demo)
#   MODEL         Gemma tag to run (default: picked from GPU memory, or RAM if there is no GPU)
#   EXTRA_MODELS  more tags to download so you can switch live, e.g. "gemma3n:e4b gemma3:4b gemma2:9b"
#   DOMAIN        your own domain with an A record to this Droplet (default: <ip>.sslip.io)
#   REPO_URL      default https://github.com/dabrara/GDG_Hackathon_10_30_26.git
#   BRANCH        default main
set -Eeuo pipefail
trap 'echo "Failed on line $LINENO. Scroll up for the error." >&2' ERR

REPO_URL="${REPO_URL:-https://github.com/dabrara/GDG_Hackathon_10_30_26.git}"
BRANCH="${BRANCH:-main}"
APP_USER_NAME="${APP_USER:-demo}"
SRC_DIR=/opt/plainsight-src
APP_DIR=/opt/plainsight
CRED_FILE=/root/plainsight-credentials.txt
TAG_RE='^gemma(2|3|3n|4)?:[a-z0-9._-]+$'
export DEBIAN_FRONTEND=noninteractive

[[ $EUID -eq 0 ]] || { echo "Run as root:  sudo bash $0"; exit 1; }
if ! grep -qiE 'ubuntu|debian' /etc/os-release 2>/dev/null; then
  echo "This script is written for Ubuntu/Debian (the default DigitalOcean image). Continuing anyway."
fi

say() { printf '\n==> %s\n' "$*"; }

# ---- password: use APP_PASS, else reuse the saved one, else make a new one
if [[ -z "${APP_PASS:-}" ]]; then
  if [[ -f "$CRED_FILE" ]]; then APP_PASS="$(sed -n 's/^password: //p' "$CRED_FILE")"; fi
  if [[ -z "${APP_PASS:-}" ]]; then APP_PASS="$(tr -dc 'A-Za-z0-9' </dev/urandom | head -c 16 || true)"; fi
fi

# ---- 1) system packages
say "Installing system packages"
apt-get update -y
apt-get install -y git curl rsync python3 python3-venv python3-pip zstd pciutils ufw \
  gpg debian-keyring debian-archive-keyring apt-transport-https ca-certificates

# ---- 2) GPU check and label
HAS_GPU=0; VRAM_GB=0
if command -v nvidia-smi >/dev/null 2>&1 && nvidia-smi -L >/dev/null 2>&1; then
  HAS_GPU=1
  VRAM_MB="$(nvidia-smi --query-gpu=memory.total --format=csv,noheader,nounits | sort -nr | head -1 | tr -d ' ')"
  VRAM_GB=$(( ${VRAM_MB:-0} / 1024 ))
  say "NVIDIA GPU found: $(nvidia-smi --query-gpu=name --format=csv,noheader | head -1) (${VRAM_GB} GB)"
  DEPLOY_LABEL="DigitalOcean GPU Droplet"
else
  echo "WARNING: no working NVIDIA driver found. On a GPU Droplet choose the 'AI/ML ready' image, which has the driver."
  echo "         Without a GPU the model runs on the CPU and will be slow."
  DEPLOY_LABEL="DigitalOcean Droplet"
fi
IP="$(curl -fsS http://169.254.169.254/metadata/v1/interfaces/public/0/ipv4/address 2>/dev/null || curl -fsS https://api.ipify.org)"
DOMAIN="${DOMAIN:-$IP.sslip.io}"   # sslip.io is a free hostname, so Caddy can get a real HTTPS certificate

# ---- 3) Ollama (listens on localhost only)
say "Installing Ollama"
command -v ollama >/dev/null || curl -fsSL https://ollama.com/install.sh | sh
systemctl enable --now ollama
for _ in $(seq 1 60); do curl -fsS http://127.0.0.1:11434/api/tags >/dev/null 2>&1 && break; sleep 1; done

# ---- 4) pull the repo (clone the first time, update after that)
say "Getting the code from $REPO_URL ($BRANCH)"
if [[ -d "$SRC_DIR/.git" ]]; then
  git -C "$SRC_DIR" fetch --depth 1 origin "$BRANCH"
  git -C "$SRC_DIR" reset --hard "origin/$BRANCH"
else
  rm -rf "$SRC_DIR"
  git clone --depth 1 --branch "$BRANCH" "$REPO_URL" "$SRC_DIR"
fi
id plainsight >/dev/null 2>&1 || useradd --system --home "$APP_DIR" --shell /usr/sbin/nologin plainsight
mkdir -p "$APP_DIR"
rsync -a --delete --exclude .git --exclude .venv --exclude .ui_cache.json "$SRC_DIR"/ "$APP_DIR"/
python3 -m venv "$APP_DIR/.venv"
"$APP_DIR/.venv/bin/pip" install -q --upgrade pip
"$APP_DIR/.venv/bin/pip" install -q -r "$APP_DIR/requirements.txt"
"$APP_DIR/.venv/bin/python" "$APP_DIR/scripts/make_samples.py" >/dev/null 2>&1 || true
chown -R plainsight:plainsight "$APP_DIR"

# ---- 5) choose and download the models
if [[ -z "${MODEL:-}" ]]; then
  if (( HAS_GPU )); then
    # Demo-friendly picks: fast on the GPU you have. Override with MODEL=gemma4:31b for the biggest.
    if   (( VRAM_GB >= 24 )); then MODEL=gemma4:12b
    elif (( VRAM_GB >= 12 )); then MODEL=gemma4:e4b
    else MODEL=gemma4:e2b; fi
  else
    MODEL="$(python3 "$APP_DIR/scripts/select_model.py" --model-only)"
  fi
fi
[[ "$MODEL" =~ $TAG_RE ]] || { echo "MODEL must be a Gemma tag such as gemma4:e4b or gemma3:12b (got: $MODEL)"; exit 1; }
say "Downloading $MODEL (the first time takes a few minutes)"
ollama pull "$MODEL"
for m in ${EXTRA_MODELS:-}; do
  [[ "$m" =~ $TAG_RE ]] || { echo "Skipping $m: not a Gemma tag"; continue; }
  say "Downloading $m"; ollama pull "$m" || echo "Could not pull $m, continuing."
done

# ---- 6) run the app as a service (localhost only)
say "Starting PlainSight"
cat > /etc/systemd/system/plainsight.service <<UNIT
[Unit]
Description=PlainSight
After=network.target ollama.service
[Service]
User=plainsight
WorkingDirectory=$APP_DIR
Environment=MODEL=$MODEL
Environment="DEPLOY_HOST=$DEPLOY_LABEL"
ExecStart=$APP_DIR/.venv/bin/uvicorn app.main:app --host 127.0.0.1 --port 8000
Restart=always
[Install]
WantedBy=multi-user.target
UNIT
systemctl daemon-reload
systemctl enable plainsight
systemctl restart plainsight

# ---- 7) Caddy: HTTPS + password (the app has no login of its own, so never expose it without this)
if ! command -v caddy >/dev/null; then
  say "Installing Caddy"
  curl -1sLf 'https://dl.cloudsmith.io/public/caddy/stable/gpg.key' | gpg --dearmor --yes -o /usr/share/keyrings/caddy-stable-archive-keyring.gpg
  curl -1sLf 'https://dl.cloudsmith.io/public/caddy/stable/debian.deb.txt' > /etc/apt/sources.list.d/caddy-stable.list
  apt-get update -y; apt-get install -y caddy
fi
HASH="$(caddy hash-password --plaintext "$APP_PASS")"
cat > /etc/caddy/Caddyfile <<CADDY
$DOMAIN {
	basicauth {
		$APP_USER_NAME $HASH
	}
	encode gzip
	reverse_proxy 127.0.0.1:8000 {
		flush_interval -1
	}
}
CADDY
systemctl enable caddy
systemctl restart caddy

# ---- 8) firewall: SSH + web only (Ollama's port 11434 stays closed)
ufw allow OpenSSH >/dev/null; ufw allow 80/tcp >/dev/null; ufw allow 443/tcp >/dev/null; ufw --force enable >/dev/null

# ---- 9) save the login and check it is alive
umask 077
printf 'url: https://%s\nuser: %s\npassword: %s\nmodel: %s\n' "$DOMAIN" "$APP_USER_NAME" "$APP_PASS" "$MODEL" > "$CRED_FILE"
for _ in $(seq 1 30); do curl -fsS http://127.0.0.1:8000/api/health >/dev/null 2>&1 && break; sleep 1; done
echo
curl -fsS http://127.0.0.1:8000/api/health || echo "(the app is still starting; check: journalctl -u plainsight -f)"
cat <<DONE


=====================================================
 PlainSight is running with $MODEL
 Open:      https://$DOMAIN
 Username:  $APP_USER_NAME
 Password:  $APP_PASS
 (also saved in $CRED_FILE)

 Logs:      journalctl -u plainsight -f
 Update:    sudo bash $(basename "$0")      (pulls the latest code)
 Switch model: sudo systemctl edit plainsight, or re-run with MODEL=...
 Billing:   destroy the Droplet when the hackathon ends.
=====================================================
DONE
